  • (Ez 10,6) בְּצַוֺּתוֹ֙ אֶת־ הָאִ֤ישׁ לְבֻֽשׁ־
3[הבדים לאמר קח אש מבינות ]לגלגל[ מבינות לכרובים ויבא ויעמד]
  • (Ez 10,6) הַבַּדִּים֙ לֵאמֹ֔ר קַ֥ח אֵשׁ֙ מִבֵּינ֣וֹת לַגַּלְגַּ֔ל מִבֵּינ֖וֹת לַכְּרוּבִ֑ים וַיָּבֹא֙ וַֽיַּעֲמֹ֔ד
4[אצל האופן וישלח הכרו]ב את ידו֯[ מבינות לכרובים אל האש אשר]
  • (Ez 10,6) אֵ֖צֶל הָאוֹפָֽן׃ (Ez 10,7) וַיִּשְׁלַח֩ הַכְּר֨וּב אֶת־ יָד֜וֹ מִבֵּינ֣וֹת לַכְּרוּבִ֗ים אֶל־ הָאֵשׁ֙ אֲשֶׁר֙
5[בינות הכרבים וי]ש֯א ויתן אל ח֯[פני לבש הבדים ויקח ויצא]
  • (Ez 10,7) בֵּינ֣וֹת הַכְּרֻבִ֔ים וַיִּשָּׂא֙ וַיִּתֵּ֔ן אֶל־ חָפְנֵ֖י לְבֻ֣שׁ הַבַּדִּ֑ים וַיִּקַּ֖ח וַיֵּצֵֽא׃
6[וירא לכרבים תב]נ֯ית ידי אדם[ תחת כנפיהם ואראה והנה ארבעה]
  • (Ez 10,8) וַיֵּרָ֖א לַכְּרֻבִ֑ים תַּבְנִית֙ יַד־ אָדָ֔ם תַּ֖חַת כַּנְפֵיהֶֽם׃ (Ez 10,9) וָאֶרְאֶ֗ה וְהִנֵּ֨ה אַרְבָּעָ֣ה
7[אופנים אצל ]הכרובים אופן[ אחד אצל הכרוב אחד ואופן אחד]
  • (Ez 10,9) אוֹפַנִּים֮ אֵ֣צֶל הַכְּרוּבִים֒ אוֹפַ֣ן אֶחָ֗ד אֵ֚צֶל הַכְּר֣וּב אֶחָ֔ד וְאוֹפַ֣ן אֶחָ֔ד
8[אצל הכרוב אחד ו]מראה האופני֯[ם כעין אבן תרשיש ומראיהם]
  • (Ez 10,9) אֵ֖צֶל הַכְּר֣וּב אֶחָ֑ד וּמַרְאֵה֙ הָא֣וֹפַנִּ֔ים כְּעֵ֖ין אֶ֥בֶן תַּרְשִֽׁישׁ׃ (Ez 10,10) וּמַ֨רְאֵיהֶ֔ם
9[דמות אחד לאר]בעתם כאשר [יהיה האופן בתוך האופן]
  • (Ez 10,10) דְּמ֥וּת אֶחָ֖ד לְאַרְבַּעְתָּ֑ם כַּאֲשֶׁ֛ר יִהְיֶ֥ה הָאוֹפַ֖ן בְּת֥וֹךְ הָאוֹפָֽן׃
10[בלכתם אל ארבעת רבעי]הם ילכו לא יסבו֯[ בלכתם כי המקום אשר יפנה]
  • (Ez 10,11) בְּלֶכְתָּ֗ם אֶל־ אַרְבַּ֤עַת רִבְעֵיהֶם֙ יֵלֵ֔כוּ לֹ֥א יִסַּ֖בּוּ בְּלֶכְתָּ֑ם כִּ֣י הַמָּק֞וֹם אֲשֶׁר־ יִפְנֶ֤ה
11[הראש אחריו ילכו לא י]ס֯בו בלכתם [וכל בשרם וגבהם וידיהם]
  • (Ez 10,11) הָרֹאשׁ֙ אַחֲרָ֣יו יֵלֵ֔כוּ לֹ֥א יִסַּ֖בּוּ בְּלֶכְתָּֽם׃ (Ez 10,12) וְכָל־ בְּשָׂרָם֙ וְגַבֵּהֶ֔ם וִֽידֵיהֶ֖ם
12[וכנפיהם והאופנים מלא]ים עינים ס֯ב֯[יב לארבעתם אופניהם לאופנים]
  • (Ez 10,12) וְכַנְפֵיהֶ֑ם וְהָאֽוֹפַנִּ֗ים מְלֵאִ֤ים עֵינַ֙יִם֙ סָבִ֔יב לְאַרְבַּעְתָּ֖ם אוֹפַנֵּיהֶֽם׃ (Ez 10,13) לָא֖וֹפַנִּ֑ים
13[להם קורא הגלגל ]באזני ואר[בעה פנים לאחד פני האחד פני הכרוב]
  • (Ez 10,13) לָהֶ֛ם קוֹרָ֥א הַגַּלְגַּ֖ל בְּאָזְנָֽי׃ (Ez 10,14) וְאַרְבָּעָ֥ה פָנִ֖ים לְאֶחָ֑ד פְּנֵ֨י הָאֶחָ֜ד פְּנֵ֣י הַכְּר֗וּב
14[ופני השני פני אדם ]והשלישי[ פני אריה והרביעי פני נשר]
  • (Ez 10,14) וּפְנֵ֤י הַשֵּׁנִי֙ פְּנֵ֣י אָדָ֔ם וְהַשְּׁלִישִׁי֙ פְּנֵ֣י אַרְיֵ֔ה וְהָרְבִיעִ֖י פְּנֵי־ נָֽשֶׁר׃
15[וירמו הכרובים היא ה]חיה א[שר ראיתי בנהר כבר ובלכת]
  • (Ez 10,15) וַיֵּרֹ֖מּוּ הַכְּרוּבִ֑ים הִ֣יא הַחַיָּ֔ה אֲשֶׁ֥ר רָאִ֖יתִי בִּֽנְהַר־ כְּבָֽר׃ (Ez 10,16) וּבְלֶ֙כֶת֙
16[הכרובים ילכו האופנים ]א֯צ֯ל[ם ובשאת הכרובים את כנפיהם]
  • (Ez 10,16) הַכְּרוּבִ֔ים יֵלְכ֥וּ הָאוֹפַנִּ֖ים אֶצְלָ֑ם וּבִשְׂאֵ֨ת הַכְּרוּבִ֜ים אֶת־ כַּנְפֵיהֶ֗ם
17[לרום מעל הארץ לא יסבו האופנים גם הם מאצלם]
  • (Ez 10,16) לָרוּם֙ מֵעַ֣ל הָאָ֔רֶץ לֹא־ יִסַּ֧בּוּ הָאוֹפַנִּ֛ים גַּם־ הֵ֖ם מֵאֶצְלָֽם׃
frg. 2
1ב֯עמדם יעמדו וברומם ירומו אותם כ֯י֯[ רוח החיה בהם ויצא]
  • (Ez 10,17) בְּעָמְדָ֣ם יַעֲמֹ֔דוּ וּבְרוֹמָ֖ם יֵר֣וֹמּוּ אוֹתָ֑ם כִּ֛י ר֥וּחַ הַחַיָּ֖ה בָּהֶֽם׃ (Ez 10,18) וַיֵּצֵא֙
2[כב]ו֯ד יהוה מעל מפתן הבית ויעמד על הכרובים ו֯[ישאו הכרובים את]
  • (Ez 10,18) כְּב֣וֹד יְהוָ֔ה מֵעַ֖ל מִפְתַּ֣ן הַבָּ֑יִת וַֽיַּעֲמֹ֖ד עַל־ הַכְּרוּבִֽים׃ (Ez 10,19) וַיִּשְׂא֣וּ הַכְּרוּבִ֣ים אֶת־
3[כנפי]ה֯ם וירומו מן הארץ לעיני בצאתם והאופנים[ לעמתם ויעמד פתח שער]
  • (Ez 10,19) כַּ֠נְפֵיהֶם וַיֵּר֨וֹמּוּ מִן־ הָאָ֤רֶץ לְעֵינַי֙ בְּצֵאתָ֔ם וְהָאֽוֹפַנִּ֖ים לְעֻמָּתָ֑ם וַֽיַּעֲמֹ֗ד פֶּ֣תַח שַׁ֤עַר
4[בית ]י֯הוה הקדמוני וכבוד א[ל]הי ישראל ע[ליהם מלמעלה היא החיה]
  • (Ez 10,19) בֵּית־ יְהוָה֙ הַקַּדְמוֹנִ֔י וּכְב֧וֹד אֱלֹהֵֽי־ יִשְׂרָאֵ֛ל עֲלֵיהֶ֖ם מִלְמָֽעְלָה׃ (Ez 10,20) הִ֣יא הַחַיָּ֗ה
5[אשר ]ר֯איתי תחת אלהי ישראל בנהר כבר֯[ ואדע כי כרובים המה]
  • (Ez 10,20) אֲשֶׁ֥ר רָאִ֛יתִי תַּ֥חַת אֱלֹהֵֽי־ יִשְׂרָאֵ֖ל בִּֽנְהַר־ כְּבָ֑ר וָאֵדַ֕ע כִּ֥י כְרוּבִ֖ים הֵֽמָּה׃
6[ארבעה ארבעה פנ]י֯ם לאחד וארבעה כנפים לא[חד ודמות ידי אדם]
  • (Ez 10,21) אַרְבָּעָ֨ה אַרְבָּעָ֤ה פָנִים֙ לְאֶחָ֔ד וְאַרְבַּ֥ע כְּנָפַ֖יִם לְאֶחָ֑ד וּדְמוּת֙ יְדֵ֣י אָדָ֔ם
7[תחת כנפיהם ודמות ]פניהם המה הפנים אשר֯[ ראיתי על נהר]
  • (Ez 10,21) תַּ֖חַת כַּנְפֵיהֶֽם׃ (Ez 10,22) וּדְמ֣וּת פְּנֵיהֶ֔ם הֵ֣מָּה הַפָּנִ֗ים אֲשֶׁ֤ר רָאִ֙יתִי֙ עַל־ נְהַר־
8[כבר מראיהם ואותם אי]ש אל עבר פניו ילכו ו֯ת֯[שא אתי רוח ותבא אתי]
  • (Ez 10,22) כְּבָ֔ר מַרְאֵיהֶ֖ם וְאוֹתָ֑ם אִ֛ישׁ אֶל־ עֵ֥בֶר פָּנָ֖יו יֵלֵֽכוּ׃ (Ez 11,1) וַתִּשָּׂ֨א אֹתִ֜י ר֗וּחַ וַתָּבֵ֣א אֹ֠תִי
9[אל שער בית יהוה ]הקדמוני הפונה קדמה֯[ והנה בפתח השער]
  • (Ez 11,1) אֶל־ שַׁ֨עַר בֵּית־ יְהוָ֤ה הַקַּדְמוֹנִי֙ הַפּוֹנֶ֣ה קָדִ֔ימָה וְהִנֵּה֙ בְּפֶ֣תַח הַשַּׁ֔עַר
10[עשרים וחמשה איש ואר]א֯ה֯ בתוכם את יאזנ֯[יה בן עזר ואת פלטיהו]
  • (Ez 11,1) עֶשְׂרִ֥ים וַחֲמִשָּׁ֖ה אִ֑ישׁ וָאֶרְאֶ֨ה בְתוֹכָ֜ם אֶת־ יַאֲזַנְיָ֧ה בֶן־ עַזֻּ֛ר וְאֶת־ פְּלַטְיָ֥הוּ
11[בן בניהו שרי העם ויאמר אלי בן]
  • (Ez 11,1) בֶן־ בְּנָיָ֖הוּ שָׂרֵ֖י הָעָֽם׃ פ (Ez 11,2) וַיֹּ֖אמֶר אֵלָ֑י בֶּן־
12[אדם אלה האנשים ה]ח֯שבים און והיעצי[ם עצת רע בעיר הזאת]
  • (Ez 11,2) אָדָ֕ם אֵ֣לֶּה הָאֲנָשִׁ֞ים הַחֹשְׁבִ֥ים אָ֛וֶן וְהַיֹּעֲצִ֥ים עֲצַת־ רָ֖ע בָּעִ֥יר הַזֹּֽאת׃
13[האמרים לא בקרוב בנות] בתים היא הסי֯[ר ואנחנו הבשר לכן הנבא]
  • (Ez 11,3) הָאֹ֣מְרִ֔ים לֹ֥א בְקָר֖וֹב בְּנ֣וֹת בָּתִּ֑ים הִ֣יא הַסִּ֔יר וַאֲנַ֖חְנוּ הַבָּשָֽׂר׃ (Ez 11,4) לָכֵ֖ן הִנָּבֵ֣א
14[עליהם הנבא בן אדם ]ותפל עלי רוח י[הוה ויאמר אלי אמר כה אמר]
  • (Ez 11,4) עֲלֵיהֶ֑ם הִנָּבֵ֖א בֶּן־ אָדָֽם׃ (Ez 11,5) וַתִּפֹּ֣ל עָלַי֮ ר֣וּחַ יְהוָה֒ וַיֹּ֣אמֶר אֵלַ֗י אֱמֹר֙ כֹּה־ אָמַ֣ר
15[יהוה כן אמרתם בית ]ישראל ומעלות[ רוחכם אני ידעתיה]
  • (Ez 11,5) יְהוָ֔ה כֵּ֥ן אֲמַרְתֶּ֖ם בֵּ֣ית יִשְׂרָאֵ֑ל וּמַעֲל֥וֹת רֽוּחֲכֶ֖ם אֲנִ֥י יְדַעְתִּֽיהָ׃
16[הרביתם חלליכם בעיר הז]את ומלאתם חו֯[צתיה חלל לכן כה אמר]
  • (Ez 11,6) הִרְבֵּיתֶ֥ם חַלְלֵיכֶ֖ם בָּעִ֣יר הַזֹּ֑את וּמִלֵּאתֶ֥ם חוּצֹתֶ֖יהָ חָלָֽל׃ פ (Ez 11,7) לָכֵ֗ן כֹּֽה־ אָמַר֮
17[אדני יהוה חלליכם אשר] שמתם בתוכה[ המה הבשר והיא]
  • (Ez 11,7) אֲדֹנָ֣י יְהוִה֒ חַלְלֵיכֶם֙ אֲשֶׁ֣ר שַׂמְתֶּ֣ם בְּתוֹכָ֔הּ הֵ֥מָּה הַבָּשָׂ֖ר וְהִ֣יא
18[הסיר ואתכם הוציא מתוכה ]חרב יראתם[ וחרב אביא עליכם]
  • (Ez 11,7) הַסִּ֑יר וְאֶתְכֶ֖ם הוֹצִ֥יא מִתּוֹכָֽהּ׃ (Ez 11,8) חֶ֖רֶב יְרֵאתֶ֑ם וְחֶ֙רֶב֙ אָבִ֣יא עֲלֵיכֶ֔ם
19[נאם אדני יהוה והוצאתי אתכ]ם֯ מתוכה [ונתתי אתכם ביד זרים]
  • (Ez 11,8) נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ (Ez 11,9) וְהוֹצֵאתִ֤י אֶתְכֶם֙ מִתּוֹכָ֔הּ וְנָתַתִּ֥י אֶתְכֶ֖ם בְּיַד־ זָרִ֑ים
20[ועשיתי בכם שפטים בחרב ]ת֯פלו ע[ל גבול ישראל אשפוט אתכם]
  • (Ez 11,9) וְעָשִׂ֛יתִי בָכֶ֖ם שְׁפָטִֽים׃ (Ez 11,10) בַּחֶ֣רֶב תִּפֹּ֔לוּ עַל־ גְּב֥וּל יִשְׂרָאֵ֖ל אֶשְׁפּ֣וֹט אֶתְכֶ֑ם
21[וידעתם כי אני יהוה ה]י֯א[ לא תהיה לכם לסיר ואתם תהיו]
  • (Ez 11,10) וִֽידַעְתֶּ֖ם כִּֽי־ אֲנִ֥י יְהוָֽה׃ (Ez 11,11) הִ֗יא לֹֽא־ תִהְיֶ֤ה לָכֶם֙ לְסִ֔יר וְאַתֶּ֛ם תִּהְי֥וּ …
frg. 3 i
